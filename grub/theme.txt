# bootique: the grub boot-menu theme. A GRUB theme (not the 16-colour
# menu_color knob) so colours are exact hex. Generated; do not hand-edit this
# deployed copy. Layout + palette:
#   - a FRAME near the screen edges (menu_pixmap_style = frame_*.png). Menu
#     items at the top, labels near the bottom.
#   - menu items deep purple #3F2470 (matches the desktop titlebar). The
#     SELECTED item is bold purple on a light-green bar (sel_*.png).
#   - labels sit over the dark foliage, so they are light: the timeout is bold
#     vibrant green, the help line lavender.
#   - item_spacing 0 so rows are adjacent (uniform, and the snapshot columns,
#     box-drawing │, form a CONTIGUOUS vertical rule).
#   - the font is GNU Unifont 16, the one built into the signed GRUB: Secure
#     Boot refuses every font file from disk (see setup.sh), so it is the only
#     font there is. The menu is made readable by RESOLUTION instead: the
#     drop-in runs it at 1024x768, which the panel scales up. Rows are 20px
#     (the original 30/24 proportion at 16px). Percent geometry is
#     resolution-independent.
desktop-image: "background.png"
title-text: ""
terminal-font: "GNU Unifont Regular 16"
# The console window for the boot messages ("Loading Linux .../initrd ..."):
# with a theme, gfxmenu defaults it to a jarring CENTRED half-screen black box.
# Full-screen makes selecting an entry read as a normal boot instead. NOTE:
# global properties need their % values QUOTED (unlike component props) or grub
# rejects the WHOLE theme ("enclose literal values in quotes") and falls back to
# the plain text menu.
terminal-left: "0%"
terminal-top: "0%"
terminal-width: "100%"
terminal-height: "100%"
